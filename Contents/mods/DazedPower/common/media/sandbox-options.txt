/* Note the path: media/sandbox-options.txt, NOT media/lua/. The B41 layout
   is silently dead in B42 -- neither of the two paths the loader checks
   reaches it, and nothing is logged when it is missing.

   One dot in the option id nests it: DazedPower.LinkRadius becomes
   SandboxVars.DazedPower.LinkRadius. Two dots would not nest and the option
   would land flat at the top level instead. */

{
    type = integer,
    min = 4,
    max = 30,
    default = 12,
    page = DazedPower,
    translation = DazedPower_LinkRadius,

/* Who may pick a placed part back up. 1 anyone, 2 anyone with a screwdriver,
   3 the person who placed it plus their safehouse plus admins.

   This exists because the mod cannot rely on the base rule: any mod that
   loosens canPickUpMoveable (Rebalanced Prop Moving is the obvious one)
   otherwise makes every panel free to take. Enum values are 1-based, and
   an enum option requires numValues > 0 AND default > 0 or it is rejected. */

option DazedPower.PickupLock
{
    type = enum,
    numValues = 3,
    default = 3,
    page = DazedPower,
    translation = DazedPower_PickupLock,
    valueTranslation = DazedPower_PickupLock,
}

/* TransformerLoss is what each powered transformer draws doing nothing, in
   watts, so a sprawling grid costs something to keep running. A transformer's
   own line runs 2.5 times LinkRadius. */

option DazedPower.TransformerLoss
{
    type = integer,
    min = 0,
    max = 500,
    default = 25,
    page = DazedPower,
    translation = DazedPower_TransformerLoss,
}

/* One residential house in this many gets a yard rig, and (about an eighth as
   often) a storage barn holds spare panels and racks, on newly generated ground
   only. 0 turns both off. */
option DazedPower.RigChance
{
    type = integer,
    min = 0,
    max = 500,
    default = 15,
    page = DazedPower,
    translation = DazedPower_RigChance,
}

/* Wiring (0.2.0). CableTilesPerWire is how many tiles one Electric Wire covers when a cable is run

option DazedPower.CableTilesPerWire
{
    type = integer,
    min = 0,
    max = 20,
    default = 4,
    page = DazedPower,
    translation = DazedPower_CableTilesPerWire,
}

option DazedPower.LiveShock
{
    type = boolean,
    default = true,

/* Climate (0.5.0). ColdStarts: a gas engine below -5 C at its own square may fail to start (60% at -25 C). */
option DazedPower.ColdStarts
{
    type = boolean,
    default = true,
    page = DazedPower,
    translation = DazedPower_ColdStarts,
